import os import requests API_URL = "https://archive.org/services/reviews.php" def post_review(identifier: str, title: str, body: str, stars: int) -> dict: if not 1 <= stars <= 5: raise ValueError("stars must be between 1 and 5") access = os.environ["IA_ACCESS_KEY"] secret = os.environ["IA_SECRET_KEY"] response = requests.post( API_URL, params={"identifier": identifier}, auth=(access, secret), data={ "identifier": identifier, "reviewtitle": title, "reviewbody": body, "stars": stars, "action": "add-review", }, timeout=15, ) response.raise_for_status() return response.json()
